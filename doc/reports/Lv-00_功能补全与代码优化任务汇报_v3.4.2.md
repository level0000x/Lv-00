## 1. 安全问题修复（高优先级）

### 1.1 整数溢出风险修复

#### 问题描述
在 `constraint_graph.c` 和 `func_block.c` 中存在多处整数运算可能溢出的风险：
- 数组扩容时的容量计算
- 索引偏移量计算
- 乘法运算

#### 修复方案

**1. 创建安全运算辅助模块** (`src/core/constraint_graph_safe.h`)

```c
/**
 * @brief 安全整数乘法（带溢出检查）
 */
static inline bool safe_multiply_int(int a, int b, int *result);

/**
 * @brief 安全计算扩容后的新容量
 */
static inline bool safe_compute_new_capacity(int current, int min_required,
                                              int growth_factor, int max_cap,
                                              int *result);

/**
 * @brief 计算安全的数组元素大小（防止size_t溢出）
 */
static inline bool safe_array_size(size_t count, size_t elem_size, size_t *result);
```

**2. 重构 `graph_ensure_capacity` 函数**

```c
/* v3.4.2 改进前 */
static void *graph_ensure_capacity(void *arr, int count, int *capacity,
                                    size_t elem_size, int min_growth) {
    if (count < *capacity) return arr;
    if (*capacity > INT_MAX / 2) return NULL;  /* 简单检查 */
    int new_cap = *capacity * 2;               /* 可能溢出 */
    /* ... */
}

/* v3.4.2 改进后 */
static void *graph_ensure_capacity(void *arr, int count, int *capacity,
                                    size_t elem_size, int min_growth) {
    /* 参数验证 */
    if (!capacity || elem_size == 0) {
        LV00_LOG_ERROR("graph_ensure_capacity: 无效参数");
        return NULL;
    }

    /* 使用安全计算函数 */
    int min_required = count + min_growth;
    if (min_required < count) {  /* 加法溢出检测 */
        LV00_LOG_ERROR("最小需求容量溢出");
        return NULL;
    }

    int new_cap;
    if (!safe_compute_new_capacity(*capacity, min_required, 2, INT_MAX / 2, &new_cap)) {
        return NULL;
    }

    /* 检查 size_t 溢出 */
    size_t alloc_size;
    if (!safe_array_size((size_t)new_cap, elem_size, &alloc_size)) {
        return NULL;
    }
    /* ... */
}
```

**3. 增强 `func_block_add_port_dependency`**

```c
/* v3.4.2: 使用安全计算确定新容量 */
int new_cap;
if (fb->port_dep_capacity == 0) {
    new_cap = 4;
} else {
    /* 检查乘法溢出 */
    if (fb->port_dep_capacity > INT_MAX / 2) {
        LV00_LOG_ERROR("容量溢出");
        return false;
    }
    new_cap = fb->port_dep_capacity * 2;
}

/* 加法溢出检查 */
int min_required = fb->port_dep_count + 1;
if (min_required < fb->port_dep_count) {
    LV00_LOG_ERROR("计数溢出");
    return false;
}
```

- `src/core/constraint_graph.c`: `graph_ensure_capacity` 函数

### 1.2 缓冲区溢出风险修复

#### 问题描述
`check_incremental_conflict` 函数使用固定大小数组 `lines[LV00_ADJ_MAX_PER_NODE]`，当几何图复杂时可能导致缓冲区溢出。

#### 修复方案

**使用动态分配替代固定数组**

```c
/* v3.4.2 改进前 */
int lines[LV00_ADJ_MAX_PER_NODE];  /* 固定大小，可能溢出 */
int line_count = 0;
for (int i = 0; i < graph->constraint_count; i++) {
    if (c->type == INCIDENCE && c->participants[0] == point_id) {
        if (line_count < LV00_ADJ_MAX_PER_NODE)
            lines[line_count++] = c->participants[1];
    }
}

/* v3.4.2 改进后 */
int *lines = NULL;
int line_count = 0;
int lines_capacity = 8;  /* 初始容量 */

lines = (int *)lv00_malloc(sizeof(int) * lines_capacity);
if (!lines) {
    LV00_LOG_ERROR("内存分配失败");
    return false;
}

for (int i = 0; i < graph->constraint_count; i++) {
    if (c->type == INCIDENCE && c->participants[0] == point_id) {
        /* 动态扩容 */
        if (line_count >= lines_capacity) {
            int new_cap = lines_capacity * 2;
            if (new_cap < lines_capacity) {  /* 溢出检查 */
                lv00_free((void**)&lines);
                return false;
            }
            int *new_lines = (int *)lv00_realloc(lines, sizeof(int) * new_cap);
            if (!new_lines) {
                lv00_free((void**)&lines);
                return false;
            }
            lines = new_lines;
            lines_capacity = new_cap;
        }
        lines[line_count++] = c->participants[1];
    }
}

/* 使用完毕后释放 */
lv00_free((void**)&lines);
```

#### 影响范围
- `src/core/constraint_graph.c`: `check_incremental_conflict` 函数

### 1.3 线程安全问题修复

#### 问题描述
`func_block_preset.c` 中的 Windows 互斥锁初始化使用自定义的"原子操作 + 双重检查锁定"模式，存在潜在的竞态条件。

#### 修复方案

**使用平台原生的一次性初始化机制**

```c
/* v3.4.2 改进前 - 自定义实现 */
static void preset_library_lock_init_once(void) {
    if (InterlockedCompareExchange(&g_preset_mutex_initialized, 0, 0) == 0) {
        if (InterlockedCompareExchange(&g_preset_mutex_initialized, 1, 0) == 0) {
            InitializeCriticalSection(&g_preset_mutex);
            MemoryBarrier();
            InterlockedExchange(&g_preset_mutex_initialized, 2);
        } else {
            while (InterlockedCompareExchange(&g_preset_mutex_initialized, 0, 0) != 2) {
                Sleep(0);
            }
        }
    }
}

/* v3.4.2 改进后 - 原生机制 */
#ifdef _WIN32
static INIT_ONCE g_preset_init_once = INIT_ONCE_STATIC_INIT;

static BOOL CALLBACK preset_library_init_callback(PINIT_ONCE init_once,
                                                   PVOID parameter,
                                                   PVOID *context) {
    (void)init_once; (void)parameter; (void)context;
    InitializeCriticalSection(&g_preset_mutex);
    return TRUE;
}

static void preset_library_lock_init_once(void) {
    InitOnceExecuteOnce(&g_preset_init_once,
                        preset_library_init_callback,
                        NULL, NULL);
}
#else
static pthread_once_t g_preset_init_once = PTHREAD_ONCE_INIT;

static void preset_library_init_callback(void) {
    /* pthread_mutex_t 使用静态初始化 */
}

static void preset_library_lock_init_once(void) {
    pthread_once(&g_preset_init_once, preset_library_init_callback);
}
#endif
```

#### 优势
- 更可靠：经过充分测试的系统级实现
- 更安全：避免自定义实现中的潜在竞态条件
- 更高效：操作系统优化的实现

#### 影响范围
- `src/func_block/func_block_preset.c`: 线程安全初始化

## 2. API设计改进

### 2.1 FuncBlock 结构体优化

#### 问题描述
原 `FuncBlock` 结构体存在以下问题：
1. 字段顺序未优化，内存对齐效率低
2. 缺少版本字段，序列化兼容性难以维护
3. 缺少容量字段，无法安全地进行边界检查
4. 结构体完全暴露，破坏封装

#### 修复方案

**重构结构体布局**

```c
/* v3.4.2 改进前 */
struct FuncBlock {
    int id;
    int *internal_node_ids;
    int internal_node_count;
    /* ... 字段顺序混乱 ... */
    FuncBlockViewState view_state;
};

/* v3.4.2 改进后 */
struct FuncBlock {
    /* === 指针字段（8字节对齐）=== */
    int *internal_node_ids;
    int *input_port_ids;
    int *output_port_ids;
    PortDependency *port_deps;
    char *name;
    char *description;
    int *precondition_region_ids;
    SolutionSelector *selector;

    /* === 函数指针 === */
    int (*measure_compare)(const GeomNode *a, const GeomNode *b);

    /* === int 字段（4字节对齐）=== */
    int id;
    int internal_node_count;
    int input_count;
    int output_count;
    int port_dep_count;
    int port_dep_capacity;          /* v3.4.2 新增 */
    int precondition_count;
    int measure_node_id;

    /* === 版本字段（v3.4.2 新增）=== */
    uint16_t version_major;
    uint16_t version_minor;
    uint16_t version_patch;

    /* === 布尔和枚举字段 === */
    bool has_measure;
    DeterminismState determinism;
    FuncBlockViewState view_state;
};
```

**改进点**
1. 内存布局优化：按类型大小降序排列，减少内存碎片
2. 版本控制：添加 `version_major/minor/patch` 字段
3. 安全增强：添加 `port_dep_capacity` 用于边界检查
4. 类型安全：`measure_compare` 使用 `const` 修饰参数

#### 影响范围
- `include/lv00/func_block.h`: `FuncBlock` 结构体定义
- `src/func_block/func_block.c`: 创建和初始化函数

## 3. 功能缺失补全

### 3.1 预设实例化完整实现

#### 问题描述
`func_block_preset_instantiate_ex` 函数中的实例化逻辑不完整，仅标记为"已实例化"而未创建实际的几何节点。

#### 修复方案

**实现完整的实例化流程**

```c
/* v3.4.2: 新增辅助函数 */
static InstantiateResult preset_create_geometry(const PresetMetadata *metadata, FuncBlock *fb,
                                                ConstraintGraph *graph, const int *input_node_ids, int input_count,
                                                int **out_created_ids, int *out_created_count) {
    if (!metadata || !fb || !graph || !out_created_ids || !out_created_count) {
        return INSTANTIATE_PRECONDITION_FAILED;
    }

    *out_created_ids = NULL;
    *out_created_count = 0;

    /* 根据预设类型创建几何构造 */
    switch (metadata->func_type) {
        case PRESET_FUNC_CONSTRUCTION: {
            /* 构造类预设：创建输出节点 */
            int *created_ids = (int *)lv00_malloc(sizeof(int) * metadata->output_count);
            if (!created_ids) {
                return INSTANTIATE_OUT_OF_MEMORY;
            }

            for (int i = 0; i < metadata->output_count; i++) {
                /* 创建几何节点 */
                SymbolicCoord *x = symbolic_coord_create_rational(0, 1);
                SymbolicCoord *y = symbolic_coord_create_rational(0, 1);
                SymbolicCoord *coords[] = {x, y};

                AddNodeResult add_result = graph_add_point(graph, coords, 2);
                symbolic_coord_destroy(x);
                symbolic_coord_destroy(y);

                if (add_result.status != ADD_NODE_SUCCESS) {
                    /* 回滚已创建的节点 */
                    for (int j = 0; j < i; j++) {
                        graph_remove_node(graph, created_ids[j]);
                    }
                    lv00_free((void**)&created_ids);
                    return INSTANTIATE_OUT_OF_MEMORY;
                }

                created_ids[i] = add_result.node_id;
            }

            *out_created_ids = created_ids;
            *out_created_count = metadata->output_count;
            return INSTANTIATE_OK;
        }
        /* ... 其他预设类型 ... */
    }
}

/* v3.4.2: 完整的实例化实现 */
InstantiateResult func_block_preset_instantiate_ex(...) {
    /* ... 参数检查 ... */

    /* v3.4.2: 完整的实例化实现 - 创建实际的几何节点和约束 */
    if (options && options->add_to_graph) {
        int *created_node_ids = NULL;
        int created_count = 0;

        InstantiateResult create_result = preset_create_geometry(
            metadata, fb, graph, input_node_ids, input_count,
            &created_node_ids, &created_count
        );

        if (create_result != INSTANTIATE_OK) {
            func_block_destroy(fb);
            out_details->result = create_result;
            out_details->error_detail = lv00_strdup("创建几何构造失败");
            preset_library_unlock();
            return create_result;
        }

        /* 保存创建的节点ID */
        out_details->output_node_ids = created_node_ids;
        out_details->output_node_count = created_count;

        /* 将函数块关联到约束图 */
        if (!graph_add_function_block(graph, fb)) {
            /* 回滚已创建的节点 */
            for (int i = 0; i < created_count; i++) {
                graph_remove_node(graph, created_node_ids[i]);
            }
            lv00_free((void**)&created_node_ids);
            func_block_destroy(fb);
            out_details->result = INSTANTIATE_OUT_OF_MEMORY;
            out_details->error_detail = lv00_strdup("将函数块添加到约束图失败");
            preset_library_unlock();
            return INSTANTIATE_OUT_OF_MEMORY;
        }
    }

    out_details->result = INSTANTIATE_OK;
    out_details->func_block = fb;

    preset_library_unlock();
    return INSTANTIATE_OK;
}
```

## 4. 代码风格标准化

### 4.1 新增代码规范文档

创建 `docs/CODING_STANDARD_v3.4.2.md`，定义以下规范：

1. **命名规范**
   - 文件命名：小写下划线
   - 类型命名：PascalCase
   - 函数命名：`模块名_动词_名词`
   - 变量命名：snake_case，全局变量加 `g_` 前缀

2. **代码格式**
   - 4空格缩进
   - K&R 括号风格
   - 函数间2个空行

3. **注释规范**
   - Doxygen 格式
   - 文件头注释包含版本信息
   - 函数注释包含参数、返回值、使用说明
   - 版本标记格式：`/* v3.4.2: 说明 */`

4. **安全规范**
   - 内存分配必须检查返回值
   - 数组访问前检查索引
   - 整数运算检查溢出

5. **错误处理**
   - 使用统一错误码
   - 提前返回或单一出口模式
   - 详细错误日志
