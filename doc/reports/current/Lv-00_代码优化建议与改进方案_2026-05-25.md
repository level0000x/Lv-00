## 📝 代码优化建议

### 1. 编译器优化提示宏

建议在 `lv00_internal.h` 中添加以下优化宏：

```c
/**
 * @file lv00_optimization.h
 * @brief 编译器优化提示宏
 * 
 * 提供分支预测提示和内联控制，帮助编译器生成更高效的代码。
 */

#ifndef LV00_OPTIMIZATION_H
#define LV00_OPTIMIZATION_H

/* ============================================================
 * 分支预测提示宏
 * ============================================================
 * 
 * 使用示例:
 *   if (LV00_LIKELY(ptr != NULL)) {
 *       // 大概率执行的代码路径
 *   }
 *   
 *   if (LV00_UNLIKELY(error != 0)) {
 *       // 小概率的错误处理路径
 *   }
 */

/**
 * @brief 提示编译器条件大概率成立
 * 
 * 用于优化分支预测，减少流水线清空。
 * 适用于正常执行路径。
 */
#if defined(__GNUC__) || defined(__clang__)
    #define LV00_LIKELY(x) __builtin_expect(!!(x), 1)
#else
    #define LV00_LIKELY(x) (x)
#endif

/**
 * @brief 提示编译器条件大概率不成立
 * 
 * 用于优化分支预测，减少流水线清空。
 * 适用于错误检查、边界条件等小概率路径。
 */
#if defined(__GNUC__) || defined(__clang__)
    #define LV00_UNLIKELY(x) __builtin_expect(!!(x), 0)
#else
    #define LV00_UNLIKELY(x) (x)
#endif

/* ============================================================
 * 函数内联控制
 * ============================================================
 */

/**
 * @brief 强制内联提示
 * 
 * 用于短小频繁调用的函数，如getter/setter。
 */
#if defined(__GNUC__) || defined(__clang__)
    #define LV00_INLINE inline __attribute__((always_inline))
#elif defined(_MSC_VER)
    #define LV00_INLINE __forceinline
#else
    #define LV00_INLINE inline
#endif

/**
 * @brief 禁止内联提示
 * 
 * 用于调试或特殊情况。
 */
#if defined(__GNUC__) || defined(__clang__)
    #define LV00_NOINLINE __attribute__((noinline))
#elif defined(_MSC_VER)
    #define LV00_NOINLINE __declspec(noinline)
#else
    #define LV00_NOINLINE
#endif

/* ============================================================
 * 内存访问优化
 * ============================================================
 */

/**
 * @brief 预取提示
 * 
 * 提示CPU提前加载数据到缓存。
 * 适用于顺序访问的数据结构遍历。
 */
#if defined(__GNUC__) || defined(__clang__)
    #define LV00_PREFETCH_READ(addr) __builtin_prefetch(addr, 0, 3)
    #define LV00_PREFETCH_WRITE(addr) __builtin_prefetch(addr, 1, 3)
#else
    #define LV00_PREFETCH_READ(addr) ((void)0)
    #define LV00_PREFETCH_WRITE(addr) ((void)0)
#endif

/**
 * @brief 内存屏障
 * 
 * 确保内存操作的顺序。
 */
#if defined(__GNUC__) || defined(__clang__)
    #define LV00_MEMORY_BARRIER() __sync_synchronize()
#elif defined(_MSC_VER)
    #include <intrin.h>
    #define LV00_MEMORY_BARRIER() _ReadWriteBarrier()
#else
    #define LV00_MEMORY_BARRIER() ((void)0)
#endif

/* ============================================================
 * 推测执行优化
 * ============================================================
 */

/**
 * @brief 推测执行检查点
 * 
 * 用于长时间计算中允许取消的检查点。
 */
#if defined(__GNUC__) || defined(__clang__)
    #define LV00_SPECULATABLE __attribute__((optimize("O3")))
#else
    #define LV00_SPECULATABLE
#endif

#endif /* LV00_OPTIMIZATION_H */
```

### 2. 改进的NULL检查宏

建议添加更安全的NULL检查：

```c
/**
 * @brief 参数验证宏
 * 
 * 使用示例:
 *   LV00_PARAM_CHECK(ctx, return LV00_ERROR_INVALID_PARAM);
 */
#define LV00_PARAM_CHECK(param, action) \
    do { \
        if (LV00_UNLIKELY((param) == NULL)) { \
            lv00_set_error(LV00_ERROR_NULL_POINTER, #param " is NULL"); \
            action; \
        } \
    } while(0)

/**
 * @brief 参数范围验证宏
 * 
 * 使用示例:
 *   LV00_RANGE_CHECK(count, 0, 100, return LV00_ERROR_OUT_OF_RANGE);
 */
#define LV00_RANGE_CHECK(value, min_val, max_val, action) \
    do { \
        if (LV00_UNLIKELY((value) < (min_val) || (value) > (max_val))) { \
            lv00_set_error(LV00_ERROR_INVALID_PARAM, \
                #value " out of range [" #min_val ", " #max_val "]"); \
            action; \
        } \
    } while(0)

/**
 * @brief 断言宏（可关闭）
 * 
 * 发布版本可关闭以提升性能。
 */
#ifndef LV00_NDEBUG
    #define LV00_ASSERT(condition) \
        do { \
            if (LV00_UNLIKELY(!(condition))) { \
                lv00_set_error(LV00_ERROR_ASSERTION_FAILED, \
                    "Assertion failed: " #condition); \
                assert(condition); \
            } \
        } while(0)
#else
    #define LV00_ASSERT(condition) ((void)0)
#endif
```

### 3. 安全的字符串操作宏

```c
/**
 * @brief 安全字符串复制宏
 * 
 * 确保始终以'\0'结尾。
 */
#define LV00_STRLCPY(dst, src, size) \
    do { \
        size_t _len = strlen(src); \
        if (_len >= (size)) _len = (size) - 1; \
        memcpy(dst, src, _len); \
        dst[_len] = '\0'; \
    } while(0)

/**
 * @brief 安全字符串拼接宏
 */
#define LV00_STRLCAT(dst, src, size) \
    do { \
        size_t _dst_len = strlen(dst); \
        size_t _src_len = strlen(src); \
        if (_dst_len >= (size)) { \
            _dst_len = (size); \
            _src_len = 0; \
        } else { \
            _src_len = (_dst_len + _src_len >= (size)) ? \
                        ((size) - _dst_len - 1) : _src_len; \
        } \
        memcpy(dst + _dst_len, src, _src_len); \
        dst[_dst_len + _src_len] = '\0'; \
    } while(0)
```

```c
    const char *file;
    int line;
    const char *function;
    Lv00ErrorCode code;
    char message[256];
} ErrorContext;

/**
 * @brief 记录错误上下文的宏
 */
#define LV00_ERROR_CTX(error_code, fmt, ...) \
    do { \
        ErrorContext ctx = { \
            .file = __FILE__, \
            .line = __LINE__, \
            .function = __func__, \
            .code = (error_code), \
        }; \
        snprintf(ctx.message, sizeof(ctx.message), fmt, ##__VA_ARGS__); \
        lv00_record_error_context(&ctx); \
    } while(0)
```

```c
/**
 * @brief 符号坐标输入验证
 */
bool symbolic_coord_validate_input(const SymbolicCoord *c) {
    if (!c) return false;
    
    switch (c->type) {
        case RATIONAL:
            if (!c->data.rational) return false;
            if (mpz_cmp_ui(mpq_denref(c->data.rational->value), 0) == 0) 
                return false;  // 分母为0
            return true;
            
        case QUADRATIC:
            // 验证判别式
            return symbolic_coord_validate_quadratic(c);
            
        case ALGEBRAIC:
            // 验证隔离区间
            return symbolic_coord_validate_isolation(c);
            
        default:
            return false;
    }
}

/**
 * @brief 约束图输入验证
 */
bool graph_validate_node(const GeomNode *node) {
    if (!node) return false;
    
    // 验证节点类型
    if (node->type < 0 || node->type >= GEOM_TYPE_COUNT) 
        return false;
    
    // 验证ID
    if (node->id < 0) return false;
    
    // 验证坐标
    if (node->symbolic_coords) {
        for (int i = 0; i < node->coord_count; i++) {
            if (!symbolic_coord_validate_input(node->symbolic_coords[i]))
                return false;
        }
    }
    
    return true;
}
```

### 2. 资源限制

```c
/**
 * @brief 增强的资源限制
 */
typedef struct {
    size_t max_memory;           // 最大内存
    size_t max_node_count;       // 最大节点数
    size_t max_constraint_count; // 最大约束数
    size_t max_depth;           // 最大递归深度
    size_t max_rewrite_steps;   // 最大重写步数
    int max_solver_iterations;   // 最大求解迭代次数
} ResourceLimits;

static ResourceLimits g_default_limits = {
    .max_memory = 1024 * 1024 * 1024,  // 1GB
    .max_node_count = 1000000,
    .max_constraint_count = 500000,
    .max_depth = 10000,
    .max_rewrite_steps = 1000000,
    .max_solver_iterations = 10000000,
};

bool check_resource_limits(const ResourceLimits *limits) {
    MemoryStats mem;
    lv00_get_memory_stats(&mem);
    
    if (mem.current_used > limits->max_memory) {
        lv00_set_error(LV00_ERROR_OUT_OF_MEMORY, 
            "Memory limit exceeded: %zu > %zu", 
            mem.current_used, limits->max_memory);
        return false;
    }
    
    return true;
}
```

### 改进的哈希查找

```c
/**
 * @brief 优化的哈希表实现
 * 
 * 使用开放寻址和线性探测。
 */
#define HASH_LOAD_FACTOR 0.75

typedef struct HashEntry {
    uint64_t key;
    void *value;
    bool occupied;
} HashEntry;

typedef struct {
    HashEntry *entries;
    size_t capacity;
    size_t count;
    size_t max_load;
    hash_equal_fn equal;
    hash_destroy_fn destroy_key;
    hash_destroy_fn destroy_value;
} HashTable;

size_t hash_table_size_hint(size_t expected_items) {
    size_t size = 16;
    while (size * HASH_LOAD_FACTOR < expected_items && size < SIZE_MAX / 2) {
        size *= 2;
    }
    return size;
}

bool hash_table_rehash(HashTable *ht, size_t new_capacity) {
    HashEntry *old_entries = ht->entries;
    size_t old_capacity = ht->capacity;
    
    ht->entries = calloc(new_capacity, sizeof(HashEntry));
    if (!ht->entries) {
        ht->entries = old_entries;
        return false;
    }
    
    ht->capacity = new_capacity;
    ht->count = 0;
    ht->max_load = (size_t)(new_capacity * HASH_LOAD_FACTOR);
    
    // 重新插入所有元素
    for (size_t i = 0; i < old_capacity; i++) {
        if (old_entries[i].occupied) {
            hash_table_insert_internal(ht, old_entries[i].key, old_entries[i].value);
        }
    }
    
    free(old_entries);
    return true;
}
```

### 1. 错误处理模式

```c
/**
 * @brief 标准错误处理模式
 * 
 * 示例：创建节点函数
 */
GeomNode *graph_create_node_safe(ConstraintGraph *graph, GeomType type) {
    // 1. 参数验证
    LV00_PARAM_CHECK(graph, return NULL);
    LV00_RANGE_CHECK(type, 0, GEOM_TYPE_COUNT - 1, return NULL);
    
    // 2. 资源检查
    if (LV00_UNLIKELY(!check_resource_limits(&g_default_limits))) {
        return NULL;
    }
    
    // 3. 实际操作
    GeomNode *node = graph_alloc_node(graph, type);
    if (LV00_UNLIKELY(!node)) {
        LV00_ERROR_CTX(LV00_ERROR_OUT_OF_MEMORY, 
            "Failed to allocate node");
        return NULL;
    }
    
    // 4. 验证结果
    if (LV00_UNLIKELY(!graph_validate_node(node))) {
        lv00_free((void **)&node);
        LV00_ERROR_CTX(LV00_ERROR_INTERNAL, 
            "Invalid node created");
        return NULL;
    }
    
    // 5. 成功日志（可选）
    LV00_LOG_DEBUG("graph", "Created node %d of type %d", 
                   node->id, node->type);
    
    return node;
}
```

### 2. 内存分配模式

```c
/**
 * @brief 安全的内存分配宏
 */
#define LV00_SAFE_MALLOC(ptr, type, count, cleanup) \
    do { \
        (ptr) = (type *)lv00_malloc(sizeof(type) * (count)); \
        if (LV00_UNLIKELY(!(ptr))) { \
            LV00_ERROR_CTX(LV00_ERROR_OUT_OF_MEMORY, \
                "Failed to allocate " #type "[%d]", count); \
            cleanup; \
            goto cleanup; \
        } \
    } while(0)

/**
 * @brief 安全的重分配宏
 */
#define LV00_SAFE_REALLOC(ptr, type, new_count, cleanup) \
    do { \
        type *_tmp = (type *)lv00_realloc((ptr), sizeof(type) * (new_count)); \
        if (LV00_UNLIKELY(!_tmp)) { \
            LV00_ERROR_CTX(LV00_ERROR_OUT_OF_MEMORY, \
                "Failed to reallocate " #type "[%d]", new_count); \
            cleanup; \
            goto cleanup; \
        } \
        (ptr) = _tmp; \
    } while(0)

/**
 * @brief 使用示例
 */
bool create_complex_structure(ComplexStruct **out) {
    ComplexStruct *s = NULL;
    NodeArray *nodes = NULL;
    
    LV00_SAFE_MALLOC(s, ComplexStruct, 1, /* nothing */);
    LV00_SAFE_MALLOC(nodes, NodeArray, 10, /* cleanup */);
    
    // ... 使用 s 和 nodes ...
    
    *out = s;
    return true;
    
cleanup:
    if (nodes) lv00_free((void **)&nodes);
    if (s) lv00_free((void **)&s);
    return false;
}
```
