# v1.9.0 微自举 A 实施计划 — lv 解析自身 .lv 文件

> **For agentic workers:** 分 6 个 Task 实现，每个 Task 独立可执行。步骤使用 checkbox (`- [ ]`) 跟踪。

**目标:** 实现完整的 .lv 文件解析管线，使 lv 系统能读取、解析、类型检查并加载 bootstrap 目录下的 .lv 规约文件。

**架构:** 自底向上构建：词法分析器 → AST 定义 → 语法解析器 → 名字解析与类型检查 → 引擎集成 → 自举验证测试。

**技术栈:** C11 + UTF-8 + 递归下降解析 + 符号表 + 约束图集成

**依赖:** lv_LANGUAGE_SPEC.md (BNF 文法)、现有 formula_parser/dsl_compiler 基础设施

- [ ] **Step 2: 实现 lexer 核心**

在 `lv_lexer.c` 中实现：

```c
#include "lv/lv_lexer.h"
#include <string.h>
#include <ctype.h>
#include <stdlib.h>

struct LvLexer {
    const char *source;
    size_t      source_len;
    size_t      pos;
    int         line;
    int         column;
    LvToken     peek_buf[3];   /* 最多 3 个 lookahead */
    int         peek_count;
};

/* 关键字查找表 */
typedef struct {
    const char *word;
    LvTokenType type;
} KeywordEntry;

static const KeywordEntry s_keywords[] = {
    {"Angle",        LV_TOKEN_KW_ANGLE},
    {"Assert",       LV_TOKEN_KW_ASSERT},
    {"Assume",       LV_TOKEN_KW_ASSUME},
    {"Axiom",        LV_TOKEN_KW_AXIOM},
    {"Bool",         LV_TOKEN_KW_BOOL},
    {"Circle",       LV_TOKEN_KW_CIRCLE},
    {"Compute",      LV_TOKEN_KW_COMPUTE},
    {"Constraint",   LV_TOKEN_KW_CONSTRAINT},
    {"Export",       LV_TOKEN_KW_EXPORT},
    {"Let",          LV_TOKEN_KW_LET},
    {"Line",         LV_TOKEN_KW_LINE},
    {"Normalize",    LV_TOKEN_KW_NORMALIZE},
    {"Point",        LV_TOKEN_KW_POINT},
    {"Polygon",      LV_TOKEN_KW_POLYGON},
    {"Proof",        LV_TOKEN_KW_PROOF},
    {"Proposition",  LV_TOKEN_KW_PROPOSITION},
    {"Prove",        LV_TOKEN_KW_PROVE},
    {"Ray",          LV_TOKEN_KW_RAY},
    {"Scalar",       LV_TOKEN_KW_SCALAR},
    {"Segment",      LV_TOKEN_KW_SEGMENT},
    {"Theorem",      LV_TOKEN_KW_THEOREM},
    {"Triangle",     LV_TOKEN_KW_TRIANGLE},
    {"and",          LV_TOKEN_KW_AND},
    {"area",         LV_TOKEN_KW_AREA},
    {"bottom",       LV_TOKEN_KW_BOTTOM},
    {"collinear",    LV_TOKEN_KW_COLLINEAR},
    {"congruent",    LV_TOKEN_KW_CONGRUENT},
    {"distance",     LV_TOKEN_KW_DISTANCE},
    {"exists",       LV_TOKEN_KW_EXISTS},
    {"false",        LV_TOKEN_KW_FALSE},
    {"forall",       LV_TOKEN_KW_FORALL},
    {"length",       LV_TOKEN_KW_LENGTH},
    {"measure",      LV_TOKEN_KW_MEASURE},
    {"module",       LV_TOKEN_KW_MODULE},
    {"not",          LV_TOKEN_KW_NOT},
    {"or",           LV_TOKEN_KW_OR},
    {"parallel",     LV_TOKEN_KW_PARALLEL},
    {"perpendicular",LV_TOKEN_KW_PERPENDICULAR},
    {"radius",       LV_TOKEN_KW_RADIUS},
    {"tangent",      LV_TOKEN_KW_TANGENT},
    {"true",         LV_TOKEN_KW_TRUE},
    /* "implies" 和 "iff" 映射为普通标识符，在 parser 层处理 */
};

static LvTokenType lookup_keyword(const char *word, size_t len) {
    for (size_t i = 0; i < sizeof(s_keywords)/sizeof(s_keywords[0]); i++) {
        if (strlen(s_keywords[i].word) == len &&
            strncmp(s_keywords[i].word, word, len) == 0) {
            return s_keywords[i].type;
        }
    }
    return LV_TOKEN_IDENTIFIER;
}

static LvToken make_token(LvLexer *lexer, LvTokenType type, size_t start_pos, size_t length) {
    LvToken tok;
    tok.type = type;
    tok.loc.offset = start_pos;
    tok.loc.line = lexer->line;   /* 简化：实际应跟踪行号 */
    tok.loc.column = lexer->column;
    tok.start = lexer->source + start_pos;
    tok.length = length;
    return tok;
}

static void skip_whitespace_and_comments(LvLexer *lexer) {
    while (lexer->pos < lexer->source_len) {
        char c = lexer->source[lexer->pos];
        if (c == ' ' || c == '\t' || c == '\r' || c == '\n') {
            if (c == '\n') { lexer->line++; lexer->column = 1; }
            else { lexer->column++; }
            lexer->pos++;
        } else if (c == '/' && lexer->pos + 1 < lexer->source_len) {
            char next = lexer->source[lexer->pos + 1];
            if (next == '/') {
                /* 行注释: // 到行尾 */
                while (lexer->pos < lexer->source_len && lexer->source[lexer->pos] != '\n')
                    lexer->pos++;
            } else if (next == '*') {
                /* 块注释: /* ... */ */
                lexer->pos += 2;
                while (lexer->pos + 1 < lexer->source_len) {
                    if (lexer->source[lexer->pos] == '*' && lexer->source[lexer->pos + 1] == '/') {
                        lexer->pos += 2;
                        break;
                    }
                    if (lexer->source[lexer->pos] == '\n') { lexer->line++; lexer->column = 1; }
                    lexer->pos++;
                }
            } else {
                break;
            }
        } else {
            break;
        }
    }
}

/* 内部: 读取并缓存一个 token */
static LvToken lex_raw(LvLexer *lexer) {
    skip_whitespace_and_comments(lexer);
    if (lexer->pos >= lexer->source_len)
        return make_token(lexer, LV_TOKEN_EOF, lexer->pos, 0);

    size_t start = lexer->pos;
    char c = lexer->source[lexer->pos];

    /* 标识符 / 关键字 */
    if (isalpha((unsigned char)c) || c == '_') {
        while (lexer->pos < lexer->source_len &&
               (isalnum((unsigned char)lexer->source[lexer->pos]) || lexer->source[lexer->pos] == '_'))
            lexer->pos++;
        size_t len = lexer->pos - start;
        LvTokenType type = lookup_keyword(lexer->source + start, len);
        return make_token(lexer, type, start, len);
    }

    /* 数字 */
    if (isdigit((unsigned char)c)) {
        while (lexer->pos < lexer->source_len && isdigit((unsigned char)lexer->source[lexer->pos]))
            lexer->pos++;
        /* 有理数: 3/4 */
        if (lexer->pos < lexer->source_len && lexer->source[lexer->pos] == '/') {
            lexer->pos++;
            if (lexer->pos < lexer->source_len && isdigit((unsigned char)lexer->source[lexer->pos])) {
                while (lexer->pos < lexer->source_len && isdigit((unsigned char)lexer->source[lexer->pos]))
                    lexer->pos++;
                return make_token(lexer, LV_TOKEN_RATIONAL, start, lexer->pos - start);
            }
            /* 不是有理数，回退: 只返回整数部分 */
            lexer->pos = start;
            while (lexer->pos < lexer->source_len && isdigit((unsigned char)lexer->source[lexer->pos]))
                lexer->pos++;
            return make_token(lexer, LV_TOKEN_INTEGER, start, lexer->pos - start);
        }
        /* 小数: 3.14 */
        if (lexer->pos < lexer->source_len && lexer->source[lexer->pos] == '.') {
            size_t dot_pos = lexer->pos;
            lexer->pos++;
            if (lexer->pos < lexer->source_len && isdigit((unsigned char)lexer->source[lexer->pos])) {
                while (lexer->pos < lexer->source_len && isdigit((unsigned char)lexer->source[lexer->pos]))
                    lexer->pos++;
                return make_token(lexer, LV_TOKEN_DECIMAL, start, lexer->pos - start);
            }
            lexer->pos = dot_pos; /* 不是小数，回退 */
        }
        return make_token(lexer, LV_TOKEN_INTEGER, start, lexer->pos - start);
    }

    /* 字符串 */
    if (c == '"') {
        lexer->pos++;
        while (lexer->pos < lexer->source_len && lexer->source[lexer->pos] != '"') {
            if (lexer->source[lexer->pos] == '\\') lexer->pos++; /* 跳过转义 */
            lexer->pos++;
        }
        if (lexer->pos < lexer->source_len) lexer->pos++; /* 闭合 " */
        return make_token(lexer, LV_TOKEN_STRING, start, lexer->pos - start);
    }

    /* 多字符运算符 */
    if (c == '-' && lexer->pos + 1 < lexer->source_len && lexer->source[lexer->pos + 1] == '>') {
        lexer->pos += 2;
        return make_token(lexer, LV_TOKEN_ARROW, start, 2);
    }
    if (c == '=' && lexer->pos + 1 < lexer->source_len && lexer->source[lexer->pos + 1] == '=') {
        lexer->pos += 2;
        return make_token(lexer, LV_TOKEN_EQEQ, start, 2);
    }
    if (c == '!' && lexer->pos + 1 < lexer->source_len && lexer->source[lexer->pos + 1] == '=') {
        lexer->pos += 2;
        return make_token(lexer, LV_TOKEN_NEQ, start, 2);
    }
    if (c == '<' && lexer->pos + 1 < lexer->source_len && lexer->source[lexer->pos + 1] == '=') {
        lexer->pos += 2;
        return make_token(lexer, LV_TOKEN_LE, start, 2);
    }
    if (c == '>' && lexer->pos + 1 < lexer->source_len && lexer->source[lexer->pos + 1] == '=') {
        lexer->pos += 2;
        return make_token(lexer, LV_TOKEN_GE, start, 2);
    }
    if ((c == '|' && lexer->pos + 1 < lexer->source_len && lexer->source[lexer->pos + 1] == '-') ||
        (c == '|' && lexer->pos + 1 < lexer->source_len && lexer->source[lexer->pos + 1] == '=')) {
        char n = lexer->source[lexer->pos + 1];
        lexer->pos += 2;
        return make_token(lexer, n == '-' ? LV_TOKEN_DARROW : LV_TOKEN_MODELS, start, 2);
    }

    /* 单字符运算符 */
    lexer->pos++;
    switch (c) {
        case '(': return make_token(lexer, LV_TOKEN_LPAREN, start, 1);
        case ')': return make_token(lexer, LV_TOKEN_RPAREN, start, 1);
        case '{': return make_token(lexer, LV_TOKEN_LBRACE, start, 1);
        case '}': return make_token(lexer, LV_TOKEN_RBRACE, start, 1);
        case '[': return make_token(lexer, LV_TOKEN_LBRACKET, start, 1);
        case ']': return make_token(lexer, LV_TOKEN_RBRACKET, start, 1);
        case ';': return make_token(lexer, LV_TOKEN_SEMICOLON, start, 1);
        case ',': return make_token(lexer, LV_TOKEN_COMMA, start, 1);
        case '.': return make_token(lexer, LV_TOKEN_DOT, start, 1);
        case ':': return make_token(lexer, LV_TOKEN_COLON, start, 1);
        case '=': return make_token(lexer, LV_TOKEN_EQUALS, start, 1);
        case '+': return make_token(lexer, LV_TOKEN_PLUS, start, 1);
        case '-': return make_token(lexer, LV_TOKEN_MINUS, start, 1);
        case '*': return make_token(lexer, LV_TOKEN_STAR, start, 1);
        case '/': return make_token(lexer, LV_TOKEN_SLASH, start, 1);
        case '^': return make_token(lexer, LV_TOKEN_CARET, start, 1);
        case '<': return make_token(lexer, LV_TOKEN_LT, start, 1);
        case '>': return make_token(lexer, LV_TOKEN_GT, start, 1);
        default:
            return make_token(lexer, LV_TOKEN_ERROR, start, 1);
    }
}

/* ── 公共 API 实现 ── */

LvLexer *lv_lexer_create(const char *source, size_t source_len) {
    LvLexer *lexer = (LvLexer *)lv_malloc(sizeof(LvLexer));
    if (!lexer) return NULL;
    lexer->source = source;
    lexer->source_len = source_len;
    lexer->pos = 0;
    lexer->line = 1;
    lexer->column = 1;
    lexer->peek_count = 0;
    return lexer;
}

void lv_lexer_destroy(LvLexer *lexer) {
    lv_free((void **)&lexer);
}

LvToken lv_lexer_next(LvLexer *lexer) {
    if (lexer->peek_count > 0) {
        LvToken tok = lexer->peek_buf[0];
        lexer->peek_count--;
        for (int i = 0; i < lexer->peek_count; i++)
            lexer->peek_buf[i] = lexer->peek_buf[i + 1];
        return tok;
    }
    return lex_raw(lexer);
}

LvToken lv_lexer_peek(LvLexer *lexer, int lookahead) {
    while (lexer->peek_count <= lookahead) {
        lexer->peek_buf[lexer->peek_count++] = lex_raw(lexer);
    }
    return lexer->peek_buf[lookahead];
}

LvSourceLoc lv_lexer_get_loc(const LvLexer *lexer) {
    LvSourceLoc loc;
    loc.line = lexer->line;
    loc.column = lexer->column;
    loc.offset = lexer->pos;
    return loc;
}

const char *lv_token_type_name(LvTokenType type) {
    static const char *names[] = {
        "INTEGER", "RATIONAL", "DECIMAL", "STRING", "IDENTIFIER",
        /* ... 所有类型名称 ... */
    };
    if (type >= 0 && type < LV_TOKEN_COUNT)
        return names[type];
    return "UNKNOWN";
}

size_t lv_token_text(const LvToken *token, char *buf, size_t buf_size) {
    if (!token || !buf || buf_size == 0) return 0;
    size_t copy_len = token->length < buf_size - 1 ? token->length : buf_size - 1;
    strncpy(buf, token->start, copy_len);
    buf[copy_len] = '\0';
    return copy_len;
}
```

```c
#include <stdio.h>
#include <string.h>
#include "lv/lv_lexer.h"

#define TEST(n) printf("  [TEST] %s ... ", n)

static void test_single_token(const char *src, LvTokenType expected_type) {
    LvLexer *lex = lv_lexer_create(src, strlen(src));
    LvToken tok = lv_lexer_next(lex);

    /* 运算符 */

    test_single_token("==", LV_TOKEN_EQEQ);
    test_single_token("!=", LV_TOKEN_NEQ);

    /* 完整语句 */

        printf("[完整语句]\n");
        const char *src = "Point A, B, C;\nConstraint collinear(A, B, C);\nProve true;";
        LvLexer *lex = lv_lexer_create(src, strlen(src));
        LvToken tok;
        int count = 0;
        LvTokenType expected[] = {
            LV_TOKEN_KW_POINT, LV_TOKEN_IDENTIFIER, LV_TOKEN_COMMA,
            LV_TOKEN_IDENTIFIER, LV_TOKEN_COMMA, LV_TOKEN_IDENTIFIER,
            LV_TOKEN_SEMICOLON,
            LV_TOKEN_KW_CONSTRAINT, LV_TOKEN_KW_COLLINEAR,
            LV_TOKEN_LPAREN, LV_TOKEN_IDENTIFIER, LV_TOKEN_COMMA,
            LV_TOKEN_IDENTIFIER, LV_TOKEN_COMMA, LV_TOKEN_IDENTIFIER,
            LV_TOKEN_RPAREN, LV_TOKEN_SEMICOLON,
            LV_TOKEN_KW_PROVE, LV_TOKEN_KW_TRUE, LV_TOKEN_SEMICOLON,
            LV_TOKEN_EOF
        };
        int expected_count = sizeof(expected) / sizeof(expected[0]);
        bool ok = true;
        while ((tok = lv_lexer_next(lex)).type != LV_TOKEN_EOF && count < expected_count) {
            if (tok.type != expected[count]) {
                printf("  token %d: expected %s, got %s\n", count,
                       lv_token_type_name(expected[count]), lv_token_type_name(tok.type));
                ok = false;
            }
            count++;
        }
        if (count != expected_count) {
            printf("  count mismatch: expected %d, got %d\n", expected_count, count);
            ok = false;
        }
        TEST("完整语句 lex");
        if (ok) PASS(); else FAIL("token 序列不匹配");
        lv_lexer_destroy(lex);
    }

    printf("\n=== %d passed, %d failed ===\n", P, F);
    return F > 0 ? 1 : 0;
}
```

```c
void lv_ast_append_child(LvAstNode *parent, LvAstNode *child) {
    if (!parent || !child) return;
    if (!parent->child) {
        parent->child = child;
    } else {
        LvAstNode *last = parent->child;
        while (last->next) last = last->next;
        last->next = child;
    }
    parent->child_count++;
}

void lv_ast_destroy(LvAstNode *node) {
    if (!node) return;
    /* 递归销毁子节点 */
    LvAstNode *cur = node->child;
    while (cur) {
        LvAstNode *next = cur->next;
        lv_ast_destroy(cur);
        cur = next;
    }
    /* 释放联合体中的数据 */
    switch (node->type) {
        case LV_AST_DECLARATION:
            lv_free((void **)&node->data.decl.names);
            break;
        case LV_AST_LET:
            lv_free((void **)&node->data.let_def.name);
            lv_free((void **)&node->data.let_def.type_name);
            break;
        case LV_AST_IDENTIFIER_EXPR:
            lv_free((void **)&node->data.ident.name);
            break;
        /* ... 其他类型 ... */
        default: break;
    }
    lv_free((void **)&node);
```

在 `lv_parser.c` 中实现解析器，遵循 `lv_LANGUAGE_SPEC.md` 的 BNF：

```c
#include "lv/lv_parser.h"
#include "lv_utils.h"
#include <string.h>
#include <stdio.h>
#include <stdarg.h>

struct LvParser {
    LvLexer *lexer;
    LvToken  current;
    int      error_count;
    LvParseError errors[64];
};

LvParser *lv_parser_create(LvLexer *lexer) {
    LvParser *parser = (LvParser *)lv_calloc(1, sizeof(LvParser));
    if (!parser) return NULL;
    parser->lexer = lexer;
    parser->current = lv_lexer_next(lexer);
    return parser;
}

void lv_parser_destroy(LvParser *parser) {
    lv_free((void **)&parser);
}

/* 辅助函数 */
static void parser_error(LvParser *parser, const char *fmt, ...) {
    if (parser->error_count >= 64) return;
    LvParseError *err = &parser->errors[parser->error_count++];
    err->loc = parser->current.loc;
    va_list args;
    va_start(args, fmt);
    vsnprintf(err->message, sizeof(err->message), fmt, args);
    va_end(args);
}

static LvToken advance(LvParser *parser) {
    parser->current = lv_lexer_next(parser->lexer);
    return parser->current;
}

static bool expect(LvParser *parser, LvTokenType type) {
    if (parser->current.type == type) {
        advance(parser);
        return true;
    }
    parser_error(parser, "期望 %s, 遇到 %s",
                 lv_token_type_name(type), lv_token_type_name(parser->current.type));
    return false;
}

/* 递归下降解析函数 */

static LvAstNode *parse_declaration(LvParser *parser, int entity_type) {
    /* DeclarationStmt ::= EntityType IdentifierList ";" */
    /* IdentifierList ::= Identifier ("," Identifier)* */
    LvSourceLoc loc = parser->current.loc;
    advance(parser); /* 消费 entity type 关键字 */

    LvAstNode *node = lv_ast_create(LV_AST_DECLARATION, loc);
    if (!node) return NULL;
    node->data.decl.entity_type = entity_type;

    /* 收集标识符 */
    char names_buf[1024];
    int pos = 0;
    while (parser->current.type == LV_TOKEN_IDENTIFIER) {
        if (pos > 0) { names_buf[pos++] = ','; }
        size_t len = parser->current.length;
        if (pos + (int)len + 1 < 1024) {
            strncpy(names_buf + pos, parser->current.start, len);
            pos += (int)len;
        }
        advance(parser);
        if (parser->current.type == LV_TOKEN_COMMA) {
            advance(parser);
        } else break;
    }
    names_buf[pos] = '\0';
    node->data.decl.names = lv_strdup(names_buf);

    expect(parser, LV_TOKEN_SEMICOLON);
    return node;
}

/* ... 更多解析函数: parse_logic_expr, parse_constraint, parse_prove 等 ... */

static LvAstNode *parse_statement(LvParser *parser) {
    switch (parser->current.type) {
        case LV_TOKEN_KW_POINT:
        case LV_TOKEN_KW_LINE:
        case LV_TOKEN_KW_CIRCLE:
        case LV_TOKEN_KW_SEGMENT:
        case LV_TOKEN_KW_TRIANGLE:
        case LV_TOKEN_KW_ANGLE:
        case LV_TOKEN_KW_POLYGON:
        case LV_TOKEN_KW_SCALAR:
        case LV_TOKEN_KW_BOOL:
        case LV_TOKEN_KW_PROPOSITION:
        case LV_TOKEN_KW_PROOF: {
            int entity_type = parser->current.type;
            return parse_declaration(parser, entity_type);
        }
        case LV_TOKEN_KW_CONSTRAINT: {
            LvSourceLoc loc = parser->current.loc;
            advance(parser);
            LvAstNode *expr = parse_logic_expr(parser);
            expect(parser, LV_TOKEN_SEMICOLON);
            LvAstNode *node = lv_ast_create(LV_AST_CONSTRAINT_STMT, loc);
            if (node) node->data.stmt.expr = expr;
            return node;
        }
        case LV_TOKEN_KW_PROVE: {
            LvSourceLoc loc = parser->current.loc;
            advance(parser);
            LvAstNode *expr = parse_logic_expr(parser);
            expect(parser, LV_TOKEN_SEMICOLON);
            LvAstNode *node = lv_ast_create(LV_AST_PROVE_STMT, loc);
            if (node) node->data.stmt.expr = expr;
            return node;
        }
        /* ... 其他语句类型 ... */
        default:
            parser_error(parser, "意外的 token: %s", lv_token_type_name(parser->current.type));
            return NULL;
    }
}

LvParseResult lv_parser_parse_program(LvParser *parser) {
    LvParseResult result = {0};
    LvSourceLoc loc = {1, 1, 0};
    LvAstNode *program = lv_ast_create(LV_AST_PROGRAM, loc);
    if (!program) { result.error_count = -1; return result; }

    while (parser->current.type != LV_TOKEN_EOF && parser->error_count < 64) {
        LvAstNode *stmt = parse_statement(parser);
        if (stmt) {
            lv_ast_append_child(program, stmt);
        } else {
            /* 出错时跳到下一个分号 */
            while (parser->current.type != LV_TOKEN_EOF && parser->current.type != LV_TOKEN_SEMICOLON)
                advance(parser);
            if (parser->current.type == LV_TOKEN_SEMICOLON)
                advance(parser);
        }
    }

    result.ast = program;
    result.error_count = parser->error_count;
    if (result.error_count > 0) {
        memcpy(result.errors, parser->errors, sizeof(LvParseError) * result.error_count);
    }
    return result;
}
```
