## 🎨 人性化设计原则

### 核心理念
1. **直觉性**: API设计符合直觉，降低学习成本
2. **一致性**: 统一的命名和调用模式
3. **容错性**: 友好的错误提示和恢复机制
4. **效率性**: 减少样板代码，提升开发效率

### 1. 链式调用模式

建议添加构建器模式的链式API：

```c
/**
 * @brief 几何对象构建器
 * 
 * 使用链式调用创建几何对象，提高可读性。
 * 
 * 使用示例:
 *   GeomNode *point = GeomBuilder.point(0, 1)
 *                                   .named("origin")
 *                                   .trusted(TRUST_GREEN)
 *                                   .build(graph);
 */
typedef struct GeomBuilder GeomBuilder;

GeomBuilder *geom_builder_new(void);
GeomBuilder *geom_builder_point(GeomBuilder *b, int64_t x_num, uint64_t x_den,
                                int64_t y_num, uint64_t y_den);
GeomBuilder *geom_builder_line(GeomBuilder *b, int p1_id, int p2_id);
GeomBuilder *geom_builder_named(GeomBuilder *b, const char *name);
GeomBuilder *geom_builder_trusted(GeomBuilder *b, TrustLevel trust);
GeomBuilder *geom_builder_label(GeomBuilder *b, const char *label);
GeomNode *geom_builder_build(GeomBuilder *b, ConstraintGraph *graph);
void geom_builder_destroy(GeomBuilder *b);
```

```c
/**
 * @brief 友好的错误信息
 */
const char *lv00_error_description(Lv00ErrorCode code) {
    switch (code) {
        case LV00_ERROR_INVALID_PARAM:
            return "参数无效：请检查输入参数是否正确";
        case LV00_ERROR_NULL_POINTER:
            return "空指针错误：请确保所有指针都已初始化";
        case LV00_ERROR_OUT_OF_MEMORY:
            return "内存不足：请尝试减少数据量或增加系统内存";
        case LV00_ERROR_NODE_CONFLICT:
            return "节点冲突：可能存在重复的几何对象";
        case LV00_ERROR_CONSTRAINT_CONFLICT:
            return "约束冲突：几何约束之间存在矛盾";
        default:
            return "未知错误";
    }
}

/**
 * @brief 获取详细的错误上下文
 */
ErrorDetails *lv00_get_error_details(void) {
    static ErrorDetails details;
    // 填充详细信息
    return &details;
}

/**
 * @brief 打印友好的错误信息
 */
void lv00_print_error友好(void) {
    Lv00ErrorCode code = lv00_get_last_error_code();
    const char *msg = lv00_get_last_error_message();
    const char *desc = lv00_error_description(code);
    
    printf("❌ 错误 [%d]: %s\n", code, desc);
    printf("   详细信息: %s\n", msg ? msg : "无");
    
    ErrorDetails *details = lv00_get_error_details();
    if (details->file) {
        printf("   位置: %s:%d (%s)\n", 
               details->file, details->line, details->function);
    }
}
```

### 2. 测试框架增强

```c
/**
 * @brief 单元测试宏
 * 
 * 简单易用的测试框架。
 */
#define LV00_TEST_SUITE(name) \
    static int test_suite_run_##name = 0; \
    static int test_suite_passed_##name = 0; \
    static void run_test_##name(const char *test_name, bool (*test_fn)(void))

#define LV00_TEST(test_name, test_body) \
    do { \
        test_suite_run_##name++; \
        printf("Running %s...", #test_name); \
        fflush(stdout); \
        if (test_body) { \
            test_suite_passed_##name++; \
            printf(" ✓\n"); \
        } else { \
            printf(" ✗\n"); \
        } \
    } while(0)

#define LV00_TEST_ASSERT(condition) \
    do { \
        if (!(condition)) { \
            printf("Assertion failed: %s at %s:%d\n", \
                   #condition, __FILE__, __LINE__); \
            return false; \
        } \
    } while(0)

#define LV00_TEST_EQUAL(a, b) \
    LV00_TEST_ASSERT((a) == (b))

#define LV00_TEST_STREQ(a, b) \
    LV00_TEST_ASSERT(strcmp((a), (b)) == 0)

#define LV00_TEST_SUMMARY(name) \
    do { \
        printf("\n=== Test Summary for %s ===\n", #name); \
        printf("Passed: %d/%d\n", test_suite_passed_##name, test_suite_run_##name); \
        printf("Result: %s\n", test_suite_passed_##name == test_suite_run_##name ? "ALL PASSED" : "SOME FAILED"); \
    } while(0)

/**
 * @brief 使用示例
 */
LV00_TEST_SUITE(geometry) {
    LV00_TEST("point_creation", {
        LV00Engine *e = lv00_engine_create();
        int p = lv00_add_point(e, 1, 1, 2, 1);
        LV00_TEST_ASSERT(p >= 0);
        lv00_engine_destroy(e);
        return p >= 0;
    });
    
    LV00_TEST("line_segment", {
        LV00Engine *e = lv00_engine_create();
        int p1 = lv00_add_point(e, 0, 1, 0, 1);
        int p2 = lv00_add_point(e, 3, 1, 0, 1);
        int s = lv00_add_line_segment(e, p1, p2);
        LV00_TEST_ASSERT(s >= 0);
        lv00_engine_destroy(e);
        return s >= 0;
    });
}

int main(void) {
    lv00_init();
    run_test_geometry();
    LV00_TEST_SUMMARY(geometry);
    lv00_cleanup();
    return 0;
}
```

### 3. 可视化调试

```c
/**
 * @brief Graphviz DOT输出
 * 
 * 生成DOT格式的图描述，用于可视化调试。
 */
bool graph_export_dot(const ConstraintGraph *graph, FILE *out) {
    fprintf(out, "digraph G {\n");
    fprintf(out, "    rankdir=LR;\n");
    fprintf(out, "    node [shape=box];\n\n");
    
    // 输出节点
    for (int i = 0; i < graph->node_count; i++) {
        GeomNode *n = graph->nodes[i];
        const char *type_name = geom_type_name(n->type);
        const char *trust_color = trust_color_name(n->trust);
        
        fprintf(out, "    n%d [label=\"%s #%d\" style=filled fillcolor=%s];\n",
                n->id, type_name, n->id, trust_color);
    }
    
    // 输出约束
    for (int i = 0; i < graph->constraint_count; i++) {
        Constraint *c = graph->constraints[i];
        const char *constraint_name = constraint_type_name(c->type);
        
        for (int j = 0; j < c->participant_count; j++) {
            fprintf(out, "    n%d -> c%d;\n", c->participants[j], c->id);
        }
        fprintf(out, "    c%d [label=\"%s\" shape=ellipse];\n",
                c->id, constraint_name);
    }
    
    fprintf(out, "}\n");
    return true;
}

/**
 * @brief JSON导出（用于Web可视化）
 */
bool graph_export_json(const ConstraintGraph *graph, FILE *out) {
    fprintf(out, "{\n");
    fprintf(out, "  \"nodes\": [\n");
    for (int i = 0; i < graph->node_count; i++) {
        GeomNode *n = graph->nodes[i];
        fprintf(out, "    {\"id\": %d, \"type\": \"%s\", \"trust\": \"%s\"}%s\n",
                n->id, geom_type_name(n->type), trust_color_name(n->trust),
                i < graph->node_count - 1 ? "," : "");
    }
    fprintf(out, "  ],\n");
    fprintf(out, "  \"constraints\": [...]\n");
    fprintf(out, "}\n");
    return true;
}
```

```c
/**
 * @brief 事务开始
 */
int graph_transaction_begin(ConstraintGraph *graph);

/**
 * @brief 事务提交
 */
int graph_transaction_commit(ConstraintGraph *graph);

/**
 * @brief 事务回滚
 */
int graph_transaction_rollback(ConstraintGraph *graph);

/**
 * @brief 使用示例
 */
void example_with_transaction(ConstraintGraph *graph) {
    graph_transaction_begin(graph);
    
    int p1 = graph_add_point(graph, coords1);
    int p2 = graph_add_point(graph, coords2);
    int seg = graph_add_line_segment(graph, p1, p2);
    
    if (solve() != 0) {
        printf("求解失败，回滚操作...\n");
        graph_transaction_rollback(graph);
        return;
    }
    
    graph_transaction_commit(graph);
    printf("所有操作成功提交！\n");
}
```
